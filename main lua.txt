function showToast(message)
  Handler(Looper.getMainLooper()).post(function()
    pcall(function()
      Toast.makeText(ctx, message, Toast.LENGTH_SHORT).show()
    end)
  end)
end

local APP_PREFS = "SquadMaster_Prefs"
local prefs = ctx.getSharedPreferences(APP_PREFS, Context.MODE_PRIVATE)

function isVibrationEnabled()
  return prefs.getBoolean("vibration", true)
end

function triggerFeedback()
  pcall(function()
    if isVibrationEnabled() and ctx then
      ctx.getSystemService(Context.VIBRATOR_SERVICE).vibrate(50)
    end
  end)
end

activeDialogs = {}

function registerDialog(dlg)
  if dlg then
    table.insert(activeDialogs, dlg)
  end
end

function dismissAllDialogs()
  for _, d in ipairs(activeDialogs) do
    pcall(function() d.dismiss() end)
  end
  activeDialogs = {}
end

function copyOnlyText(text)
  pcall(function()
    triggerFeedback()
    local clipboard = ctx.getSystemService(Context.CLIPBOARD_SERVICE)
    local clip = ClipData.newPlainText("label", text)
    clipboard.setPrimaryClip(clip)
    showToast("Copied to Clipboard!")
  end)
end

function shareTextMenu(text)
  pcall(function()
    triggerFeedback()
    dismissAllDialogs()
    Handler(Looper.getMainLooper()).postDelayed(Runnable({
      run = function()
        pcall(function()
          local intent = Intent(Intent.ACTION_SEND)
          intent.setType("text/plain")
          intent.putExtra(Intent.EXTRA_TEXT, text)
          local chooser = Intent.createChooser(intent, "Share Squad via...")
          chooser.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
          ctx.startActivity(chooser)
        end)
      end
    }), 150)
  end)
end

function openWhatsApp(number, msg)
  pcall(function()
    dismissAllDialogs()
    local textStr = msg or "Assalam-o-Alaikum! I am using the Squad Master extension."
    local intent = Intent(Intent.ACTION_VIEW, Uri.parse("whatsapp://send?phone=" .. number .. "&text=" .. Uri.encode(textStr)))
    intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
    ctx.startActivity(intent)
  end)
end

function openYouTubeChannel(urlStr)
  pcall(function()
    dismissAllDialogs()
    local intent = Intent(Intent.ACTION_VIEW, Uri.parse(urlStr))
    intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
    ctx.startActivity(intent)
  end)
end

function openWhatsAppLink(url)
    pcall(function()
        dismissAllDialogs()
        local options = {"WhatsApp Messenger", "WhatsApp Business"}
        local pDlg = LuaDialog(ctx)
        registerDialog(pDlg)
        pDlg.setTitle("Choose WhatsApp")
        local l = LinearLayout(ctx)
        l.setOrientation(1)
        l.setPadding(30, 30, 30, 30)
        for idx, optName in ipairs(options) do
            local btn = Button(ctx)
            btn.setText(optName)
            btn.onClick = function()
                triggerFeedback()
                pDlg.dismiss()
                local pkgName = (idx == 1) and "com.whatsapp" or "com.whatsapp.w4b"
                pcall(function()
                    local intent = Intent(Intent.ACTION_VIEW, Uri.parse(url))
                    intent.setPackage(pkgName)
                    intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
                    ctx.startActivity(intent)
                end)
            end
            l.addView(btn)
        end
        pDlg.setView(l)
        pDlg.show()
    end)
end

function showExitConfirmation(parentDlg)
  triggerFeedback()
  local exitDlg = LuaDialog(ctx)
  registerDialog(exitDlg)
  exitDlg.setTitle("Exit Confirmation")
  
  local exitLay = LinearLayout(ctx)
  exitLay.setOrientation(1)
  exitLay.setPadding(30, 30, 30, 30)
  
  local exitMsg = TextView(ctx)
  exitMsg.setText("Are you sure you want to close this extension?")
  exitMsg.setTextSize(16)
  exitMsg.setPadding(0, 0, 0, 20)
  exitLay.addView(exitMsg)
  
  local btnRow = LinearLayout(ctx)
  btnRow.setOrientation(0)
  
  local yesBtn = Button(ctx)
  yesBtn.setText("Yes")
  yesBtn.setLayoutParams(LinearLayout.LayoutParams(0, -2, 1))
  yesBtn.onClick = function()
    triggerFeedback()
    dismissAllDialogs()
    if parentDlg then
      pcall(function() parentDlg.dismiss() end)
    end
    pcall(function()
      if ctx.stopSelf then
        ctx.stopSelf()
      elseif ctx.finish then
        ctx.finish()
      end
    end)
  end
  
  local noBtn = Button(ctx)
  noBtn.setText("No")
  noBtn.setLayoutParams(LinearLayout.LayoutParams(0, -2, 1))
  noBtn.onClick = function()
    triggerFeedback()
    exitDlg.dismiss()
  end
  
  btnRow.addView(yesBtn)
  btnRow.addView(noBtn)
  exitLay.addView(btnRow)
  exitDlg.setView(exitLay)
  exitDlg.show()
end

function showMainMenu()
  local dlg = LuaDialog(ctx)
  registerDialog(dlg)
  dlg.setTitle("Squad Master")
  local mainLay = LinearLayout(ctx)
  mainLay.setOrientation(1)
  mainLay.setPadding(30, 30, 30, 30)

  local greetingTv = TextView(ctx)
  greetingTv.setText("Welcome Squad Master")
  greetingTv.setTextSize(18)
  greetingTv.setTextColor(0xFF00796B)
  greetingTv.setTypeface(nil, 1)
  greetingTv.setGravity(Gravity.CENTER)
  greetingTv.setPadding(0, 0, 0, 5)
  mainLay.addView(greetingTv)

  local headerTv = TextView(ctx)
  headerTv.setText("Developed by rizvi Lover cricket board")
  headerTv.setTextSize(14)
  headerTv.setTextColor(0xFF555555)
  headerTv.setTypeface(nil, 1)
  headerTv.setGravity(Gravity.CENTER)
  headerTv.setPadding(0, 0, 0, 20)
  mainLay.addView(headerTv)

  local b2 = Button(ctx)
  b2.setText("Squad Manager")
  b2.onClick = function()
    if showTeamSquadOptions then
      showTeamSquadOptions()
    else
      showToast("Squad Master module not loaded!")
    end
  end
  mainLay.addView(b2)
  
  local bAi = Button(ctx)
  bAi.setText("AI Assistant")
  bAi.onClick = function()
    if openAiWeb then
      openAiWeb()
    else
      showToast("AI Assistant module not loaded!")
    end
  end
  mainLay.addView(bAi)

  -- یہاں Settings کا بٹن پہلے کر دیا گیا ہے
  local bSettings = Button(ctx)
  bSettings.setText("Settings")
  bSettings.onClick = function()
    if showSettings then
      showSettings()
    else
      showToast("Settings module not loaded!")
    end
  end
  mainLay.addView(bSettings)

  -- یہاں About کا بٹن اس کی جگہ پر کر دیا گیا ہے
  local bAbout = Button(ctx)
  bAbout.setText("About")
  bAbout.onClick = function()
    if showAboutDialog then
      showAboutDialog()
    else
      showToast("About module not loaded!")
    end
  end
  mainLay.addView(bAbout)

  local b4 = Button(ctx)
  b4.setText("Exit")
  b4.onClick = function() 
    showExitConfirmation(dlg)
  end
  mainLay.addView(b4)

  dlg.setView(mainLay)
  dlg.setCancelable(false)
  dlg.setOnKeyListener(function(dialog, keyCode, event)
    if keyCode == KeyEvent.KEYCODE_BACK and event.getAction() == KeyEvent.ACTION_UP then
      showExitConfirmation(dlg)
      return true
    end
    return false
  end)

  dlg.show()
end

showMainMenu()
