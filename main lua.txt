local function openLink(url)
  pcall(function()
    local intent = Intent(Intent.ACTION_VIEW, Uri.parse(url))
    intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
    ctx.startActivity(intent)
  end)
end

local function sortData()
  table.sort(savedData, function(a, b)
    local t1 = a.title and tostring(a.title):lower() or ""
    local t2 = b.title and tostring(b.title):lower() or ""
    return t1 < t2
  end)
end

local function saveData()
  sortData()
  pcall(function()
    local f = io.open(dataFile, "w")
    if f then
      for _, item in ipairs(savedData) do
        local safe_content = tostring(item.content):gsub("\n", "<NEWLINE>")
        f:write(tostring(item.title) .. "|_|_|" .. safe_content .. "\n")
      end
      f:close()
    else
      showToast("Error: Data could not be saved!")
    end
  end)
end

local function loadData()
  savedData = {}
  pcall(function()
    local f = io.open(dataFile, "r")
    if f then
      for line in f:lines() do
        local t, c = line:match("^(.*)|_|_|(.*)$")
        if t and c then
          c = c:gsub("\r", "")
          local original_content = c:gsub("<NEWLINE>", "\n")
          table.insert(savedData, {title = t, content = original_content})
        end
      end
      f:close()
    end
  end)
  sortData()
end

local function performExport(itemsToExport)
  if #itemsToExport == 0 then
    showToast("No items to export!")
    return
  end
  
  local timeStr = os.date("%Y%m%d_%H%M%S")
  local exportPath = "/storage/emulated/0/Download/DataSaver_Backup_" .. timeStr .. ".txt"
  
  pcall(function()
    local f = io.open(exportPath, "w")
    if f then
      for _, item in ipairs(itemsToExport) do
        local safe_title = tostring(item.title):gsub("\n", " ") 
        local safe_content = tostring(item.content):gsub("\n", "<NEWLINE>")
        f:write(safe_title .. "|_|_|" .. safe_content .. "\n")
      end
      f:close()
      
      local success_dlg = LuaDialog()
      success_dlg.setTitle("Exported Successfully")
      success_dlg.setMessage(#itemsToExport .. " item(s) have been successfully backed up to one single file at:\n\n" .. exportPath)

      success_dlg.show()
      
      speakTTS("Data Exported Successfully")
    else
      showToast("Export Failed! Please check storage permissions.")
    end
  end)
end

local function importFile(filePath)
  pcall(function()
    local f = io.open(filePath, "r")
    local importedCount = 0
    if f then
      for line in f:lines() do
        local t, c = line:match("^(.*)|_|_|(.*)$")
        if t and c then
          c = c:gsub("\r", "")
          local original_content = c:gsub("<NEWLINE>", "\n")
          local exists = false
          for _, existing in ipairs(savedData) do
            if existing.title == t and existing.content == original_content then
              exists = true
              break
            end
          end
          if not exists then
            table.insert(savedData, {title = t, content = original_content})
            importedCount = importedCount + 1
          end
        end
      end
      f:close()
      
      if importedCount > 0 then
        saveData() 
        rebuildList(tostring(main_ids.search_bar.getText()))
        local s_dlg = LuaDialog()
        s_dlg.setTitle("Import Successful")
        s_dlg.setMessage(importedCount .. " item(s) have been imported successfully!")
        s_dlg.setButton("OK", function() s_dlg.dismiss() end)
        s_dlg.show()
        speakTTS("Import Successful")
      else
        showToast("Error: Invalid file format or data already exists!")
      end
    else
      showToast("Failed to open the file!")
    end
  end)
end

local function showCustomFileBrowser(startPath)
  local browser_dlg = LuaDialog()
  local b_ids = {}
  
  local browser_layout = {
    LinearLayout;
    orientation = "vertical";
    {
      TextView;
      id = "current_path_tv";
      text = startPath;
      padding = "10dp";
      textSize = "12sp";
      textColor = "#0055A4";
      backgroundColor = "#E0E0E0";
      layout_width = "fill";
    };
    {
      ScrollView;
      layout_width = "fill";
      layout_weight = "1";
      {
        LinearLayout;
        orientation = "vertical";
        id = "files_container";
        layout_width = "fill";
        padding = "10dp";
      };
    };
  }
  
  browser_dlg.setView(loadlayout(browser_layout, b_ids))
  browser_dlg.setTitle("Select Backup (.txt) File")
  
  local function loadPath(path)
    b_ids.files_container.removeAllViews()
    b_ids.current_path_tv.setText("Path: " .. path)
    
    local dir = File(path)
    
    if path ~= "/storage/emulated/0" and path ~= "/" then
      local backBtn = loadlayout({
        Button;
        text = "鐚拑绗� GO BACK";
        layout_width = "fill";
        textColor = "#AA0000";
      })
      pcall(function() backBtn.setAllCaps(false) end)
      backBtn.onClick = function() loadPath(dir.getParent()) end
      b_ids.files_container.addView(backBtn)
    end
    
    if dir.exists() and dir.isDirectory() then
      local files = dir.listFiles()
      if files then
        local sortedFiles = {}
        for i=0, #files-1 do table.insert(sortedFiles, files[i]) end
        table.sort(sortedFiles, function(a,b)
          if a.isDirectory() and not b.isDirectory() then return true end
          if not a.isDirectory() and b.isDirectory() then return false end
          return a.getName():lower() < b.getName():lower()
        end)
        
        for _, f in ipairs(sortedFiles) do
          local fname = tostring(f.getName())
          if f.isDirectory() and not fname:match("^%.") then
            local btn = loadlayout({
              Button;
              text = "棣冩惂 " .. fname;
              layout_width = "fill";
              textColor = "#0055A4";
            })
            pcall(function() btn.setAllCaps(false) end)
            btn.onClick = function() loadPath(f.getAbsolutePath()) end
            b_ids.files_container.addView(btn)
            
          elseif f.isFile() and fname:match("%.txt$") then 
            local btn = loadlayout({
              Button;
              text = "棣冩惈 " .. fname;
              layout_width = "fill";
              textColor = "#006600";
            })
            pcall(function() btn.setAllCaps(false) end)
            btn.onClick = function()
              browser_dlg.dismiss()
              importFile(f.getAbsolutePath())
            end
            b_ids.files_container.addView(btn)
          end
        end
      end
    else
      b_ids.files_container.addView(loadlayout({
        TextView;
        text = "This folder is empty or access is denied.";
        gravity = "center";
      }))
    end
  end
  
  loadPath(startPath)
  browser_dlg.setButton("Cancel", function() browser_dlg.dismiss() end)
  browser_dlg.show()
end

          savedData[index].content = c
          saveData()
          rebuildList(tostring(main_ids.search_bar.getText()))
          showToast("Item Updated Successfully")
          edit_dlg.dismiss()
        end;
      };
    };
  }
  
  edit_dlg.setTitle("Edit Item")
  edit_dlg.setView(loadlayout(edit_layout, ids))

    onTextChanged=function()
      local t = tostring(ids.edit_title_field.getText())
      local c = tostring(ids.edit_content_field.getText())
      
      local t_len = (utf8 and utf8.len(t)) or #t
      local c_len = (utf8 and utf8.len(c)) or #c
      
      ids.title_counter.setText(t_len .. " / " .. max_title_chars)

        if service then
          local root = service.getRootInActiveWindow()
          if root then
            local focusedNode = root.findFocus(AccessibilityNodeInfo.FOCUS_INPUT)
            if focusedNode and (focusedNode.isEditable() or (focusedNode.getClassName() and tostring(focusedNode.getClassName()) == "android.widget.EditText")) then
              pasted = focusedNode.performAction(AccessibilityNodeInfo.ACTION_PASTE)
            end
            
            if not pasted then
              local function searchAndPaste(node)
                if not node then return false end
                if node.isEditable() or (node.getClassName() and tostring(node.getClassName()) == "android.widget.EditText") then
                  node.performAction(AccessibilityNodeInfo.ACTION_FOCUS) 
                  return node.performAction(AccessibilityNodeInfo.ACTION_PASTE) 
                end
                
                for i = 0, node.getChildCount() - 1 do
                  if searchAndPaste(node.getChild(i)) then
                    return true
                  end
                end
                return false
              end
              
              pasted = searchAndPaste(root)
            end
          end
        end
      end)
      
      if pasted then
        vibrateDevice()
        speakTTS("Item Pasted")
        showToast("Content Pasted Successfully!")
      else
        speakTTS("Text Box Not Found")
        showToast("Content copied, but Text Box Not Found!")
      end
    end, 500)
  end
  
  ids.myBtn.onLongClick = function()
    local opt_dlg = LuaDialog()
    opt_dlg.setTitle("Options: " .. tostring(item_title))
    
    local opt_layout = {
      LinearLayout;
      orientation = "vertical";
      padding = "20dp";
      {
        Button;
        text = "Copy Content Only";
        layout_width = "fill";
        layout_marginBottom = "10dp";
        onClick = function()
          pcall(function()
            local clipboard = ctx.getSystemService(Context.CLIPBOARD_SERVICE)
            local clip = ClipData.newPlainText("Rizvi", tostring(item_content))
            clipboard.setPrimaryClip(clip)
          end)
          vibrateDevice()
          showToast("Content Copied to Clipboard!")
          speakTTS("Text Copied")
          opt_dlg.dismiss()
        end;
      };
      {
        Button;
        text = "Share Content";
        layout_width = "fill";
        layout_marginBottom = "10dp";
        onClick = function()
          pcall(function()
             local shareIntent = Intent(Intent.ACTION_SEND)
             shareIntent.setType("text/plain")
             shareIntent.putExtra(Intent.EXTRA_TEXT, tostring(item_content))
             ctx.startActivity(Intent.createChooser(shareIntent, "Share via..."))
          end)
          opt_dlg.dismiss()
          dlg.dismiss()
        end;
      };
      {
        Button;
        text = "Edit Item";
        layout_width = "fill";
        layout_marginBottom = "10dp";
        onClick = function()
          opt_dlg.dismiss()
          showEditDialog(orig_index)
        end;
      };
      {
        Button;
        text = "Delete Item";
        textColor = "#FF0000"; 
        layout_width = "fill";
        layout_marginBottom = "10dp";
        onClick = function()
          opt_dlg.dismiss()
          local del_dlg = LuaDialog()
          del_dlg.setTitle("Confirm Delete")
          del_dlg.setMessage("Are you sure you want to delete '" .. tostring(item_title) .. "'?")
          del_dlg.setButton("Yes, Delete", function()
            table.remove(savedData, orig_index) 
            saveData() 
            rebuildList(tostring(main_ids.search_bar.getText())) 
            speakTTS("Item Deleted")
            showToast("Item Deleted!")
          end)
          del_dlg.setButton2("Cancel", function() del_dlg.dismiss() end)
          del_dlg.show()
        end;
      };
      {
        Button;
        text = "Cancel";
        layout_width = "fill";
        onClick = function()
          opt_dlg.dismiss()
        end;
      };
    }
    
    opt_dlg.setView(loadlayout(opt_layout))
    opt_dlg.show()
    return true 
  end
  
  main_ids.dynamic_list.addView(view)
end

function rebuildList(query)
  main_ids.dynamic_list.removeAllViews()
  query = query and query:lower() or ""
  
  main_ids.total_items_count.setText("Total Saved Items: " .. #savedData)
  
  local count = 0
  for orig_index, item in ipairs(savedData) do
    if query == "" or item.title:lower():find(query, 1, true) or item.content:lower():find(query, 1, true) then
      addTitleButton(item.title, item.content, orig_index)
      count = count + 1
    end
  end
  
  if count == 0 then
    local empty_layout = {
      LinearLayout;
      layout_width = "fill";
      padding = "20dp";
      gravity = "center";
      {
        TextView;
        text = "No Data Found";
        textSize = "16sp";
        textColor = "#888888"; 
      };
    }
    main_ids.dynamic_list.addView(loadlayout(empty_layout))
  end
end
